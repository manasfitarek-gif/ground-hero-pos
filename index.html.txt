    function calculateSessionMetrics(station) {
      if (!station.timeBlocks || station.timeBlocks.length === 0) {
        return { formattedTime: '00:00:00', totalSeconds: 0, currentPrice: 0, blocksDetails: [] };
      }

      const now = station.status === 'paused' ? station.pausedTime : Date.now();
      let totalSeconds = 0;
      let currentPrice = 0;
      let blocksDetails = [];

      station.timeBlocks.forEach((block, index) => {
        const isCurrentBlock = (index === station.timeBlocks.length - 1) && station.status !== 'paused';
        const blockEnd = (isCurrentBlock || !block.endTime) ? now : block.endTime;
        const blockDurationMs = Math.max(0, blockEnd - block.startTime);
        const blockSecs = Math.floor(blockDurationMs / 1000);

        totalSeconds += blockSecs;
        const rate = HOURLY_RATES[block.joysticks] || HOURLY_RATES[1];
        const blockCost = (blockSecs / 3600) * rate;
        currentPrice += blockCost;

        const bHrs = Math.floor(blockSecs / 3600);
        const bMins = Math.floor((blockSecs % 3600) / 60);
        const bSecs = blockSecs % 60;
        const durStr = `${bHrs > 0 ? bHrs + 'h ' : ''}${bMins}m ${bSecs}s`;

        blocksDetails.push({
          joysticks: block.joysticks,
          durationFormatted: durStr,
          cost: blockCost
        });
      });

      const hrs = Math.floor(totalSeconds / 3600);
      const mins = Math.floor((totalSeconds % 3600) / 60);
      const secs = totalSeconds % 60;
      const formattedTime = `${String(hrs).padStart(2, '0')}:${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;

      return { formattedTime, totalSeconds, currentPrice, blocksDetails };
    }

    function confirmChangeJoysticks() {
      const station = stations.find(s => s.id === currentEditJoyStationId);
      if (station) {
        const now = Date.now();
        if (!station.timeBlocks) station.timeBlocks = [];

        if (station.timeBlocks.length === 0) {
          station.timeBlocks.push({
            joysticks: selectedEditJoysticksCount,
            startTime: now,
            endTime: null
          });
        } else {
          const currentBlock = station.timeBlocks[station.timeBlocks.length - 1];
          currentBlock.endTime = now;

          if (currentBlock.joysticks !== selectedEditJoysticksCount) {
            station.timeBlocks.push({
              joysticks: selectedEditJoysticksCount,
              startTime: now,
              endTime: null
            });
          }
        }

        station.joysticks = selectedEditJoysticksCount;
        saveStations();
        renderDashboard();
      }
      closeChangeJoysticksModal();
    }

      const station = stations.find(s => s.id === stationId);
      if (station && station.status === 'active') {
        const now = Date.now();
        station.status = 'paused';
        station.pausedTime = now;
        if (station.timeBlocks && station.timeBlocks.length > 0) {
          station.timeBlocks[station.timeBlocks.length - 1].endTime = now;
        }
        saveStations();
        renderDashboard();
      }
    }

    function resumeSession(stationId) {
      const station = stations.find(s => s.id === stationId);
      if (station && station.status === 'paused') {
        const now = Date.now();
        station.status = 'active';
        station.pausedTime = null;
        if (!station.timeBlocks) station.timeBlocks = [];
        station.timeBlocks.push({
          joysticks: station.joysticks,
          startTime: now,
          endTime: null
        });
        saveStations();
        renderDashboard();
      }
    }

    // CHECKOUT FLOW & CURRENCY & CHANGE CALCULATOR
    function openCheckoutModal(stationId) {
      currentCheckoutStationId = stationId;
      const station = stations.find(s => s.id === stationId);
      if (!station) return;

      const now = Date.now();
      if (station.status === 'active' && station.timeBlocks && station.timeBlocks.length > 0) {
        station.timeBlocks[station.timeBlocks.length - 1].endTime = now;
      }

      currentCheckoutCurrency = 'LBP';
      document.getElementById('checkout-currency-select').value = 'LBP';
      document.getElementById('amount-received-input').value = '';

      refreshCheckoutModalDisplay();
      document.getElementById('modal-checkout').classList.remove('hidden');
    }

    function changeCheckoutCurrency() {
      currentCheckoutCurrency = document.getElementById('checkout-currency-select').value;
      document.getElementById('amount-received-input').value = '';
      refreshCheckoutModalDisplay();
    }

    function refreshCheckoutModalDisplay() {
      const station = stations.find(s => s.id === currentCheckoutStationId);
      if (!station) return;

      const metrics = calculateSessionMetrics(station);
      const ordersTotal = (station.orders || []).reduce((sum, o) => sum + (o.price * o.qty), 0);
      const grandTotal = metrics.currentPrice + ordersTotal;

      document.getElementById('checkout-station').innerText = station.name;
      document.getElementById('checkout-duration').innerText = metrics.formattedTime;
      document.getElementById('checkout-time-price').innerText = formatCheckoutCurrency(metrics.currentPrice);
      document.getElementById('checkout-orders-price').innerText = formatCheckoutCurrency(ordersTotal);
      document.getElementById('checkout-total').innerText = formatCheckoutCurrency(grandTotal);

      document.getElementById('received-label').innerText = `Amount Received (${currentCheckoutCurrency === 'USD' ? 'USD $' : 'L.L'}):`;
      document.getElementById('amount-received-input').placeholder = currentCheckoutCurrency === 'USD' ? 'e.g. 10' : 'e.g. 500000';

      const blocksContainer = document.getElementById('checkout-blocks-list');
      blocksContainer.innerHTML = '';
      metrics.blocksDetails.forEach((block, idx) => {
        const div = document.createElement('div');
        div.className = 'flex justify-between text-slate-300';
        div.innerHTML = `<span>Block ${idx + 1}: ${block.joysticks} Joystick(s) (${block.durationFormatted})</span><span class="text-indigo-400 font-mono">${formatCheckoutCurrency(block.cost)}</span>`;
        blocksContainer.appendChild(div);
      });

      const listContainer = document.getElementById('checkout-orders-list');
      listContainer.innerHTML = '';
      if (station.orders && station.orders.length > 0) {
        station.orders.forEach(o => {
          const div = document.createElement('div');
          div.className = 'flex justify-between';
          div.innerHTML = `<span>${o.qty}x ${o.name}</span><span>${formatCheckoutCurrency(o.price * o.qty)}</span>`;
          listContainer.appendChild(div);
        });
      } else {
        listContainer.innerHTML = '<span class="italic text-slate-500">No extra orders</span>';
      }

      calculateChange();
    }

    function calculateChange() {
      const station = stations.find(s => s.id === currentCheckoutStationId);
      if (!station) return;

      const metrics = calculateSessionMetrics(station);
      const ordersTotal = (station.orders || []).reduce((sum, o) => sum + (o.price * o.qty), 0);
      const grandTotalLL = metrics.currentPrice + ordersTotal;

      const receivedInput = parseFloat(document.getElementById('amount-received-input').value);
      const changeDisplay = document.getElementById('change-due-display');

      if (isNaN(receivedInput) || receivedInput <= 0) {
        changeDisplay.innerText = currentCheckoutCurrency === 'USD' ? '$0.00' : '0 L.L';
        return;
      }

      if (currentCheckoutCurrency === 'USD') {
        const totalDueUSD = grandTotalLL / USD_TO_LBP_RATE;
        const changeUSD = receivedInput - totalDueUSD;
        if (changeUSD < 0) {
          changeDisplay.innerText = `Insufficient ($${Math.abs(changeUSD).toFixed(2)} short)`;
          changeDisplay.className = 'font-mono font-bold text-rose-400 text-base';
        } else {
          const changeLL = changeUSD * USD_TO_LBP_RATE;
          changeDisplay.innerText = `$${changeUSD.toFixed(2)} (${formatLL(changeLL)})`;
          changeDisplay.className = 'font-mono font-bold text-emerald-400 text-base';
        }
      } else {
        const changeLL = receivedInput - grandTotalLL;
        if (changeLL < 0) {
          changeDisplay.innerText = `Insufficient (${formatLL(Math.abs(changeLL))} short)`;
          changeDisplay.className = 'font-mono font-bold text-rose-400 text-base';
        } else {
          const changeUSD = changeLL / USD_TO_LBP_RATE;
          changeDisplay.innerText = `${formatLL(changeLL)} ($${changeUSD.toFixed(2)})`;
          changeDisplay.className = 'font-mono font-bold text-emerald-400 text-base';
        }
      }
    }

    function closeCheckoutModal() {
      const station = stations.find(s => s.id === currentCheckoutStationId);
      if (station && station.status === 'active' && station.timeBlocks && station.timeBlocks.length > 0) {
        station.timeBlocks[station.timeBlocks.length - 1].endTime = null;
      }
      document.getElementById('modal-checkout').classList.add('hidden');
    }

    function confirmPayment() {
      const station = stations.find(s => s.id === currentCheckoutStationId);
      if (station) {
        const metrics = calculateSessionMetrics(station);
        const snacksTotal = (station.orders || []).reduce((sum, o) => sum + (o.price * o.qty), 0);
        const gamingCost = Math.round(metrics.currentPrice);
        const totalBill = Math.round(gamingCost + snacksTotal);

        const startTime = (station.timeBlocks && station.timeBlocks.length > 0) ? station.timeBlocks[0].startTime : Date.now();

        completedSessions.unshift({
          id: Date.now(),
          stationName: station.name,
          timeBlocks: station.timeBlocks || [],
          startTime: new Date(startTime).toISOString(),
          endTime: new Date().toISOString(),
          durationFormatted: metrics.formattedTime,
          gamingCost: gamingCost,
          snacksTotal: snacksTotal,
          orders: station.orders || [],
          totalBill: totalBill
        });

        saveHistory();

        station.status = 'available';
        station.joysticks = 1;
        station.timeBlocks = [];
        station.pausedTime = null;
        station.orders = [];
        saveStations();

        renderDashboard();
      }
      closeCheckoutModal();
    }

    // END OF DAY SHEET VIEW
    function renderEOD() {
      const tbody = document.getElementById('eod-table-body');
      tbody.innerHTML = '';

      const today = new Date().toDateString();
      const todaySessions = completedSessions.filter(s => new Date(s.endTime).toDateString() === today);

      let totalGaming = 0;
      let totalSnacks = 0;

      todaySessions.forEach(s => {
        totalGaming += s.gamingCost;
        totalSnacks += s.snacksTotal;

        const orderSummary = (s.orders || []).map(o => `${o.qty}x ${o.name}`).join(', ') || 'None';
        const startFormatted = new Date(s.startTime).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
        const endFormatted = new Date(s.endTime).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

        const blocksSummary = (s.timeBlocks || []).map(b => `${b.joysticks}J (${Math.round((b.endTime - b.startTime)/60000)}m)`).join(' ➔ ');

        const tr = document.createElement('tr');
        tr.className = 'hover:bg-slate-800/50 transition-all';
        tr.innerHTML = `
          <td class="px-4 py-3 font-semibold text-white">${s.stationName}</td>
          <td class="px-4 py-3 text-slate-400 text-xs">${startFormatted} - ${endFormatted}</td>
          <td class="px-4 py-3 font-mono text-slate-300">${s.durationFormatted}</td>
          <td class="px-4 py-3 text-xs text-indigo-300 font-mono">${blocksSummary}</td>
          <td class="px-4 py-3 font-mono">${formatLL(s.gamingCost)}</td>
          <td class="px-4 py-3 text-xs text-slate-400 max-w-xs truncate" title="${orderSummary}">${orderSummary}</td>
          <td class="px-4 py-3 font-mono font-bold text-emerald-400">${formatLL(s.totalBill)}</td>
        `;
        tbody.appendChild(tr);
      });

      document.getElementById('eod-gaming-revenue').innerText = formatLL(totalGaming);
      document.getElementById('eod-snacks-revenue').innerText = formatLL(totalSnacks);
      document.getElementById('eod-total-revenue').innerText = formatLL(totalGaming + totalSnacks);
    }

    function deleteCurrentDay() {
      const today = new Date().toDateString();
      const todaySessions = completedSessions.filter(s => new Date(s.endTime).toDateString() === today);

      if (todaySessions.length === 0) {
        alert("There are no closed sessions recorded for today to delete.");
        return;
      }

      if (confirm("Are you sure you want to delete today's entire day sheet (الجرد اليومي)?")) {
        completedSessions = completedSessions.filter(s => new Date(s.endTime).toDateString() !== today);
        saveHistory();
        renderEOD();
        renderDashboard();
      }
    }

    function exportEODExcel() {
      const today = new Date().toDateString();
      const todaySessions = completedSessions.filter(s => new Date(s.endTime).toDateString() === today);

      if (todaySessions.length === 0) {
        alert("No closed sessions available for today to export.");
        return;
      }

      let csvRows = [];
      csvRows.push(["Station", "Start Time", "End Time", "Duration", "Controller Timeline", "Gaming Cost (LL)", "Orders Summary", "Orders Total (LL)", "Total Bill (LL)"].join(","));

      todaySessions.forEach(s => {
        const startFormatted = new Date(s.startTime).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
        const endFormatted = new Date(s.endTime).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
        const ordersStr = (s.orders || []).map(o => `${o.qty}x ${o.name}`).join('; ');
        const blocksStr = (s.timeBlocks || []).map(b => `${b.joysticks} Joy`).join(' -> ');

        const row = [
          `"${s.stationName}"`,
          `"${startFormatted}"`,
          `"${endFormatted}"`,
          `"${s.durationFormatted}"`,
          `"${blocksStr}"`,
          s.gamingCost,
          `"${ordersStr}"`,
          s.snacksTotal,
          s.totalBill
        ];
        csvRows.push(row.join(","));
      });

      const blob = new Blob(["\uFEFF" + csvRows.join("\r\n")], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.setAttribute("href", url);
      link.setAttribute("download", `Ground_Hero_Jard_${new Date().toISOString().slice(0, 10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }
